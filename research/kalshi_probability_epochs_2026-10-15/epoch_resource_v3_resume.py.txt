def summary(values):
    ordered=sorted(values)
    return {'n':len(values),'mean':statistics.mean(values),'p95':ordered[math.ceil(.95*len(ordered))-1],'max':max(values),'raw':values}
def measure(fn,n=20):
    values=[]
    for _ in range(n):
        start=time.thread_time_ns();fn();values.append((time.thread_time_ns()-start)/1e6)
    return summary(values)
def window(start,index):
    return Window(('KXBTC15M-'+str(index)).ljust(128,'X'),start,start+900,100,venue_start_price=100,fee_rounding='kalshi',venue='kalshi')
def fill(learner,base):
    def advance(at,windows):
        learner.batch(('batch',(learner.batch_seq or 0)+1,at,0,0,[]),at);learner.batch_complete()
        learner.advance(windows,100,.001,at,at)
    for index in range(64):
        w=window(base+1000+1800*index,base+index)
        learner.discover([w],w.start-10);advance(w.start+300,[w])
        learner.labels_received([(w.slug,bool(index%2))],w.end);advance(w.end,[w])
    starts=base+1000+1800*64
    pending=[window(starts+i/8,base+100+i) for i in range(32)]
    learner.discover(pending,starts-10);advance(starts+304,pending)
    assert len(learner.pending)==32 and len(learner.recent)==len(learner.labels)==64
    assert learner.counts['forecasts']==96 and learner.counts['scored']==64

def saved(host):
    old,envelope=host.checkpoint()

def maxhost():
    with patch.object(probability,'fingerprint',return_value='a'*64):old=probability.ProbabilityLearning(0)
    fill(old,0);state={epochs.LEGACY:old.state()}
    actual_source=probability.fingerprint()

        base=epoch*200000
        host=make(actual_source if epoch==4 else str(epoch)*64,base)
        host.restore(state,base)
        assert host.error is None
        fill(host.active,base)
        state=saved(host)
    assert state[epochs.LEGACY]['counts']['scored']==64
    assert len(state[epochs.EPOCHS]['archived'])==4
    return host,state

def trial(module,path):
    clock=[900.];view=module.WireView();writes=[]
    def sink(path,data,durable=False):writes.append((Path(path).name,durable))
    with patch.object(module,'_write',sink),patch('os.urandom',lambda n:b'x'*n):
        town=module.KalshiArena(module.ArenaConfig(path,population=12),lambda:view,clock=lambda:clock[0],cpu=lambda:0.)
        windows=[window(1000,1),window(1900,2)]
        town.receive(('batch',1,900.,0,0,[('w',[astuple(w) for w in windows]),('m',100,900.,.001)]),view);town.step(900.)
        clock[0]=1100.;town.receive(('batch',2,1100.,0,0,[('m',100,1100.,.001)]),view);town.step(1100.)
        before=time.thread_time_ns()
        for i in range(100):
            now=1100.1+i/10;clock[0]=now
            town.receive(('batch',i+3,now,0,0,[('m',100,now,.001)]),view);town.step(now)
        elapsed=(time.thread_time_ns()-before)/1e6/100
    return elapsed,writes
