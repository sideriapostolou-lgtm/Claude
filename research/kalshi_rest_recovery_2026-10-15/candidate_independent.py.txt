from types import SimpleNamespace
from nightcrawler.speeddesk_kalshi_ws import KalshiMarketWS,MarketState
from nightcrawler.weatherwatch_markets import WeatherWS
T='KXBTC15M-26OCT151545-45'
for cls in (KalshiMarketWS,WeatherWS):
    out=[]
    extra={'on_snapshot':lambda _:None} if cls is WeatherWS else {}
    ws=cls(SimpleNamespace(),tickers=lambda:frozenset({T}),on_book=lambda _:None,on_trade=lambda _:None,on_status=out.append,clock=lambda:103.,**extra)
    ws.connected=True
    ws.state=MarketState(frozenset({T}))
    ws.state.book_at[T]=ws.state.exchange_at[T]=100.
    ws.state.trade_at[T]=103.
    calls=[]
    def moving_clock():
        calls.append(1)
        return 103. if len(calls)==1 else 103.01
    ws._clock=moving_clock
    assert ws.fresh_maker_windows()=={T}
    assert len(calls)==1, 'one coherent current clock per accessor'
    assert not ws.fresh_maker_windows(), 'next sample expires old book even with fresh print'
    assert len(calls)==2
    # At first captured now, a future book cannot create makers outside ws_books.
    ws.state.book_at[T]=ws.state.exchange_at[T]=ws.state.trade_at[T]=103.01
    calls.clear()
    ws._report()
    assert len(calls)==1
    assert not out[-1]['ws_books'] and not out[-1]['ws_maker_windows']
    ws._report()
    assert len(calls)==2
    assert out[-1]['ws_books']==[T] and out[-1]['ws_maker_windows']==[T]
    # One coherent report after expiry emits a genuine invalidation.
    ws._clock=lambda:106.02
    ws._report()
    assert not out[-1]['ws_books'] and not out[-1]['ws_maker_windows']
    assert out[-1]['gap_windows']==[T]
    print(cls.__name__+' independent clock/report checks passed')
