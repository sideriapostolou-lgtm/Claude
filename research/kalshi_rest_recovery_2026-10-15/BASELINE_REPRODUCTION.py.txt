Harness first attempt preserved in tool output: after three reproduced races, fake cancel lacked keyword reason; corrected fixture only.
Source inspected: app 8c8681a; checkout HEAD 5d19e7a (docs). Source hashes printed.
Not proof that this race caused historic production gaps/losses or that fixing it

  Book errors: speeddesk_kalshi.py:285-287, including malformed responses raised

  fresh_books at281-284. Keep request clocks, poll budgets and pagination atomicity.

  deriving current fresh_books plus 0 <= now-trade_at <= STALE_S; no cached list.
  The bounds reject NaN/infinity and reversed time. fresh_books:317-321 checks

  with missing books; _cancel_unsafe can cancel resting bids on stale-feed reason.
  It does NOT itself confirm real-order cancellation or release live reservations.
  Live twin/markout invalidation:1073-1082. Arena gap:1565-1584 preserves exclusion.

"""
import asyncio
import hashlib
import json
import sys
import threading
from dataclasses import dataclass, field
from pathlib import Path
from types import SimpleNamespace

ROOT = Path('/workspace/nightcrawler-probability-learning')
sys.path.insert(0, str(ROOT / 'src'))
from nightcrawler.speeddesk_kalshi import KalshiFeeds
from nightcrawler.speeddesk_kalshi_ws import KalshiMarketWS, MarketState
from nightcrawler.speeddesk_model import Book, Level, Window
from nightcrawler.speeddesk import SpeedRunner

TICKER = 'KXBTC15M-26OCT151545-45'
START = 1800000000.0

@dataclass
class Snapshot:
    books: dict = field(default_factory=dict)
    maker_windows: frozenset = frozenset()

async def scenario(kind, action, outcome='error'):
    clock = [START + 1]
    started = asyncio.Event()
    release = threading.Event()
    loop = asyncio.get_running_loop()
    statuses, received, cancellations = [], [], []
    settings = SimpleNamespace(speeddesk_kalshi_ws_enabled=False)
    model = SimpleNamespace(books={}, cancel=lambda *args, **kwargs: cancellations.append((args,kwargs)) or [])
    runner = SimpleNamespace(
        evidence=SimpleNamespace(last_fresh={TICKER}), feed_status={'coinbase':'connected','trade_stream':True},
        _live_market_snapshot=Snapshot(), desk=model, clock=SimpleNamespace(now=lambda:clock[0]),
        decisions=lambda _:None, _refresh_live_snapshot=lambda:None, _live_call=lambda *_:None, arena=None)
    def report(value):
        statuses.append(dict(value))
        SpeedRunner.on_status(runner, {'coinbase':'connected', **value})
    def book(value):
        received.append(value)
        model.books[(value.window,value.side)] = value
    def get(_path):
        loop.call_soon_threadsafe(started.set)
        assert release.wait(5), 'fake was not released'
        if outcome == 'error':
            raise TimeoutError('synthetic failure')
        if outcome == 'invalid':
            return {}
        return {'orderbook_fp':{'yes_dollars':[['0.40','2.00']], 'no_dollars':[['0.42','2.00']]}}
    feed = KalshiFeeds(settings, on_tick=lambda *_:None, on_book=book, on_trade=lambda _:None,
        on_windows=lambda _:None,on_status=report,clock=lambda:clock[0],rest_get=get,websocket=False)
    window = Window(TICKER,START,START+900,80000,venue='kalshi',venue_start_price=80000)
    feed.windows[TICKER] = window
    ws = KalshiMarketWS(settings,tickers=lambda:frozenset(feed.windows),on_book=feed._ws_book,
        on_trade=lambda _:None,on_status=feed._ws_report,clock=lambda:clock[0])
    feed._ws = None if action == 'no_ws' else ws
    task = asyncio.create_task(feed.poll_book() if kind == 'book' else feed.poll_trades())
    await asyncio.wait_for(started.wait(),2)
    if action in ('recover','stale_cache','stale_trade','book_only'):
        clock[0] += .1
        ws.state = MarketState(frozenset({TICKER}))
        ws.connected=True
        ws.state.book_at[TICKER]=clock[0]
        ws.state.exchange_at[TICKER]=clock[0]
        if action != 'book_only':
            ws.state.trade_at[TICKER]=clock[0]
        ws._report()
        for side in ('up','down'):
            feed._ws_book(Book(TICKER,side,(Level(.4,2),),(Level(.6,2),),clock[0],'recovered',clock[0]))
        if action in ('stale_cache','stale_trade'):
            clock[0] += 3.01 # Do not run WS timeout reporting before pending REST resumes.
            if action == 'stale_trade':
                ws.state.book_at[TICKER]=clock[0]
                ws.state.exchange_at[TICKER]=clock[0]
    if action == 'removed':
        feed.windows={}
    if action == 'readmitted':
        feed.windows={}
        feed.windows[TICKER]=Window(TICKER,START,START+900,80000,venue='kalshi',venue_start_price=80000)
    before = len(model.books)
    cached = list(feed._ws_status.get('ws_maker_windows',[]))
    fresh = sorted(ws.fresh_books())
    release.set()
    assert await asyncio.wait_for(task,2)
    result = {'kind':kind,'action':action,'outcome':outcome,'books_before':before,'books_after':len(model.books),
        'fresh_ws_at_completion':fresh,'cached_ws_makers_at_completion':cached,
        'last_status':statuses[-1] if statuses else None,'received_sources':[v.source for v in received],
        'cancellations':len(cancellations)}
    if action=='recover' and outcome in ('error','invalid'):
        assert fresh==[TICKER] and before==2 and len(model.books)==0
        assert statuses[-1]['gap_windows']==[TICKER] and statuses[-1]['trade_stream'] is True
    if action=='stale_cache':
        assert not fresh and cached==[TICKER] and statuses[-1]['gap_windows']==[TICKER]
    if action=='stale_trade':
        assert fresh==[TICKER] and cached==[TICKER] and clock[0]-ws.state.trade_at[TICKER]>3
    if action=='failure_first':
        assert statuses[-1]['gap_windows']==[TICKER]
        original_gap=dict(statuses[-1])
        ws.connected=True
        ws.state=MarketState(frozenset({TICKER}))
        ws.state.book_at[TICKER]=clock[0]
        ws.state.exchange_at[TICKER]=clock[0]
        ws.state.trade_at[TICKER]=clock[0]
        ws._report()
        assert original_gap['gap_windows']==[TICKER] and not statuses[-1]['gap_windows']
        result['later_recovery_preserves_original_gap']=True
    if action=='readmitted':
        assert len(received)==2 and all(v.source=='rest' for v in received)
    return result

async def main():
    cases=[]
    for kind,action,outcome in [
        ('book','recover','error'),('book','recover','invalid'),('tape','recover','error'),
        ('book','none','error'),('book','stale_cache','error'),('tape','book_only','error'),
        ('book','removed','error'),('book','readmitted','valid'),
        ('book','no_ws','error'),('tape','stale_trade','error'),('book','failure_first','error')]:
        cases.append(await scenario(kind,action,outcome))
    paths=['src/nightcrawler/speeddesk_kalshi.py','src/nightcrawler/speeddesk_kalshi_ws.py',
           'src/nightcrawler/speeddesk.py','src/nightcrawler/speeddesk_live.py','src/nightcrawler/kalshi_arena.py']
    print(json.dumps({'source_sha256':{p:hashlib.sha256((ROOT/p).read_bytes()).hexdigest() for p in paths},
        'cases':cases,'historical_loss_causation':'unproved','app_tests_rerun':False},indent=2))

asyncio.run(main())
